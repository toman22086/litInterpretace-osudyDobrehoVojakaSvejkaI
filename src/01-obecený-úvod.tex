\section{Obecný úvod k~interpretaci}

\subsection{Základní údaje}
\begin{itemize}
    \item \textbf{Název díla:} Osudy dobrého vojáka Švejak za světové války: 1.~díl -- V~zázemí
    \item \textbf{Autor:} Jaroslav Hašek (1883-1923)
    \item \textbf{Překladatel:} Dílo pochází od českého autora, nepřekládá se.
\end{itemize}

\subsection{Osobnost a~tvorba autora, zasazení}
\begin{itemize}
    \item \textbf{O~autorovi}\cite{wiki:Jaroslav_Hasek}\cite[62]{didaktis:literatura3}
    \begin{itemize}
        \item Vystudoval Českoslovanskou obchodní akademii v~Praze
        \item Působil jako prozaik, básník, publicista, propagandista a voják
        \item Psal do časopisů Svět zvířat, Ženský obzor a~Českém slově
        \item Byl ovlivněn anarchismem a~v~roce 1911 založil Stranu mírného pokroku v~mezích zákona
    \end{itemize}
    \item \textbf{Nejznámější literární díla:} Osudy dobrého vojáka Švejak za světové války, Můj obchod se psy a~jiné humoresky, Májové výkřiky
    \item \textbf{Zasazení autora do literárního období:} Česká literatura počátku 20. století. Je představitelem modrnisku a expresionismu.
\end{itemize}
